\begin{frame}{O que é o Ligador (Linker)?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Definição}
        Programa que \textbf{combina} múltiplos arquivos-objeto e bibliotecas em um \textbf{único executável}.
      \end{block}
      \vspace{0.4cm}
      \begin{description}[\textbf{Saída:}]
        \item[\textbf{Entrada:}] arquivos \texttt{.o} + bibliotecas
        \item[\textbf{Processo:}] resolução de símbolos + relocação
        \item[\textbf{Saída:}] executável binário
      \end{description}
      \vspace{0.2cm}
      {\small\color{gray} Exemplos: \texttt{ld} (GNU), \texttt{link.exe} (MSVC)}
    \end{column}
    \begin{column}{0.48\textwidth}

\begin{frame}{Bibliotecas}
  \begin{block}{Definição}
    Coleções de \textbf{código pré-compilado} empacotadas para \textbf{reutilização}.
  \end{block}
  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{exampleblock}{Biblioteca Estática}
        \begin{itemize}
          \item Extensão: \texttt{.a} / \texttt{.lib}
          \item Código \textbf{copiado} para o executável
          \item Criada com \texttt{ar rcs}
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{Biblioteca Dinâmica}
        \begin{itemize}
          \item Extensão: \texttt{.so} / \texttt{.dll}
          \item Carregada em \textbf{runtime}
          \item \textbf{Compartilhada} entre programas
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ligação Estática}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Definição}
        Processo em que todo o código das bibliotecas é \textbf{incorporado} no executável em \textbf{tempo de compilação}.
      \end{block}
      \vspace{0.4cm}
      \cmark \; Executável \textbf{autossuficiente}\\[0.25cm]
      \cmark \; Não depende de \texttt{.so}/\texttt{.dll}\\[0.25cm]
      \cmark \; Arquivo \textbf{maior}, mas \textbf{independente}
    \end{column}
    \begin{column}{0.47\textwidth}

  \vspace{0.3cm}
  \begin{enumerate}
    \item \textbf{Resolução de Símbolos} — associa cada referência à sua definição
    \item \textbf{Relocação} — ajusta endereços para o espaço de memória final
    \item \textbf{Geração} — mescla seções \texttt{.text}, \texttt{.data}, \texttt{.bss} no executável
  \end{enumerate}
\end{frame}

\begin{frame}{Resolução de Símbolos}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{block}{Como funciona?}
        \begin{itemize}
          \item Cada \texttt{.o} tem uma \textbf{tabela de símbolos}
          \item Símbolos \textbf{definidos} — funções/variáveis locais
          \item Símbolos \textbf{indefinidos} — referências externas
          \item Ligador \textbf{casa} referências $\leftrightarrow$ definições
        \end{itemize}
      \end{block}
      \vspace{0.3cm}
      \begin{alertblock}{Erro comum}
        \texttt{undefined reference to `func'}
      \end{alertblock}
    \end{column}
    \begin{column}{0.51\textwidth}
      \begin{center}
        \begin{tikzpicture}[
          filebox/.style={draw=primary!30, fill=lightbg, rounded corners=3pt, minimum width=3.5cm, minimum height=1.6cm, align=center, font=\scriptsize},

        \end{tikzpicture}

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Relocação de Endereços}
  \begin{block}{Problema}
    Cada \texttt{.o} assume endereço inicial \textbf{0x0}.
    Quando combinados, os endereços \textbf{colidem}.
  \end{block}
  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{exampleblock}{Antes \normalfont(endereços relativos)}
        \texttt{main.o} \; $\rightarrow$ \texttt{.text} em \texttt{0x0000}\\
        \texttt{math.o} \; $\rightarrow$ \texttt{.text} em \texttt{0x0000}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Depois \normalfont(endereços absolutos)}
        \texttt{main.o} \; $\rightarrow$ \texttt{.text} em \texttt{0x1000}\\
        \texttt{math.o} \; $\rightarrow$ \texttt{.text} em \texttt{0x2000}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{center}
    O ligador \alert{recalcula} todos os endereços para posições finais no executável.
  \end{center}
\end{frame}

\begin{frame}{Geração do Executável}
  \begin{columns}[T]
    \begin{column}{0.53\textwidth}
      \begin{block}{Arquivo gerado}
        Formato \textbf{ELF} (Linux) ou \textbf{PE} (Windows):
      \end{block}
      \vspace{0.2cm}
      \begin{itemize}
        \item \textbf{\texttt{.text}} — código de máquina
        \item \textbf{\texttt{.data}} — variáveis inicializadas
        \item \textbf{\texttt{.bss}} — variáveis não inicializadas
        \item \textbf{Header} — metadados, entry point
        \item \textbf{Symbol Table} — para depuração
      \end{itemize}
    \end{column}

\section{Vantagens e Desvantagens}
% ============================================================

\begin{frame}{Ligação Estática — Prós e Contras}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{exampleblock}{Vantagens}
        \begin{itemize}
          \item \textbf{Portabilidade} — sem dependências
          \item \textbf{Desempenho} — sem overhead dinâmico
          \item \textbf{Confiabilidade} — imune ao DLL Hell
          \item \textbf{Distribuição} simples
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{Desvantagens}
        \begin{itemize}
          \item \textbf{Tamanho} do executável maior
          \item \textbf{Duplicação} de código em RAM
          \item \textbf{Atualização} — recompilar tudo
          \item \textbf{Memória} — consumo maior
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{center}
    {\small\color{gray} Casos de uso: sistemas embarcados, CLIs, containers Docker, binários Go/Rust}
  \end{center}
\end{frame}

\begin{frame}[fragile]{Código Fonte: C (Interface)}

#include <stdio.h>

int main() {
    char texto[] = "Software Basico";
    long long len = my_strlen(texto);
    printf("Tamanho: %lld\n", len);
    my_reverse(texto);
    printf("Invertido: %s\n", texto);

}
        \end{lstlisting}
      \end{block}

\begin{frame}[fragile]{Código Fonte: Assembly x86\_64}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{my\_strlen.s}}
        \begin{lstlisting}[language=C]
.globl my_strlen
my_strlen:
    movq $0, %rax
.loop:
    cmpb $0, (%rdi, %rax)
    je .end
    incq %rax
    jmp .loop
.end: ret
        \end{lstlisting}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{my\_reverse.s} (Trecho)}
        \begin{lstlisting}[language=C]
.globl my_reverse
my_reverse:
    movq %rdi, %rsi
.find_end:
    cmpb $0, (%rsi)
    je .found_end
    incq %rsi
    jmp .find_end
.found_end:
    decq %rsi
# ... (swap loop e ret)
        \end{lstlisting}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Processo de Ligação Estática}
  \begin{block}{1. Compilacao e Criacao da Biblioteca (\texttt{.a})}
    \begin{lstlisting}[language=bash]
$ as my_strlen.s -o my_strlen.o
$ as my_reverse.s -o my_reverse.o
$ ar rcs libstrutils.a my_strlen.o my_reverse.o
    \end{lstlisting}
  \end{block}

  \begin{block}{2. Ligação Estática com o Programa C}
    \begin{lstlisting}[language=bash]
$ gcc -c main.c -o main.o
$ gcc main.o -L. -lstrutils -static -o programa
    \end{lstlisting}
  \end{block}

  \vspace{0.2cm}
  \begin{center}
    Resultado: \texttt{programa} — executável \alert{100\% independente}.
  \end{center}
\end{frame}
